\begin{abstract}
Sharpness-Aware Minimization (SAM) improves deep learning generalization by minimizing the worst-case loss in a local ball $\min_w \max_{\|\epsilon\|\le\rho} L(w+\epsilon)$, seeking flat minima. In this work, we study the inverted, optimistic dual formulation---termed \textbf{Anti-SAM}:
$$\min_{w \in \mathbb{R}^d} \left( \min_{\|\epsilon\| \le \rho} L(w + \epsilon) \right)$$
which implements a morphological erosion operator down the loss landscape. We show that while Anti-SAM accelerates training loss descent by a linear velocity term $\mathcal{O}(\rho \|\nabla L(w)\|)$, it exhibits a catastrophic generalization failure, rapidly diverging on validation risk.

We mathematically uncover the geometric origins of this failure through two theorems:
(1) \textbf{The Phase-Space Caustic Collapse Theorem}: the Anti-SAM perturbation field $E(w) = -\rho \frac{\nabla L}{\|\nabla L\|}$ possesses an inherently negative divergence $\operatorname{div}(E) = -\frac{\rho}{\|\nabla L\|} \operatorname{Tr}_{T_w^\perp}(\nabla^2 L) < 0$. In overparameterized regimes with positive transverse curvature, phase-space volume contracts exponentially ($\det(J_\Phi) \to 0$), collapsing parameters into zero-measure sample-specific sharp needles whose catchment basin volume is dilated by $(\rho/r)^d$;
(2) \textbf{The Noise Divergence Gap}: finite-sample gradient noise creates an $\mathcal{O}\left(\frac{\rho d \sigma^2}{\|\nabla L_{\mathcal{D}}\|}\right)$ gap between empirical loss reduction and population risk.

To solve this dilemma, we introduce \textbf{IsoAntiSAM}, a framework built on two novel principles:
\begin{enumerate}
  \item \textbf{The Isochoric Gauge Condition}, which constrains the perturbation flow to be divergence-free on transverse submanifolds ($\operatorname{div}_{T^\perp}(E_{\text{iso}}) = 0$), strictly conserving phase-space volume and eliminating needle singularities;
  \item \textbf{The Bilateral Coherence Gate}, which computes cross-sample manifold alignment across independent micro-batches $B_1, B_2$, proving that $\mathbb{E}[\langle g_1, g_2 \rangle] = \|\nabla L_{\mathcal{D}}\|^2 \ge 0$, exactly canceling sample noise and vetoing spurious needle trajectories.
\end{enumerate}
Crucially, all foundational theoretical claims and noise cancellation theorems are formally machine-checked in the \textbf{Lean 4} theorem prover. We provide a fused HIP C++ kernel optimized for AMD Instinct MI300X accelerators and confirm synchronous validation descent.
\end{abstract}

\section{Introduction}
Modern deep learning optimization is intimately coupled to the geometry of the empirical loss surface. Standard optimizers like Stochastic Gradient Descent (SGD) and Adam minimize empirical risk $L_S(w) = \frac{1}{|S|}\sum_{z \in S} \ell(w; z)$. Foret et al. (2020) proposed Sharpness-Aware Minimization (SAM), defined as:
\begin{equation}
\min_{w \in \mathbb{R}^d} \max_{\|\epsilon\| \le \rho} L(w + \epsilon)
\end{equation}
By penalizing the supremum within a ball of radius $\rho$, SAM drives optimization toward wide, flat basins, which correlate with improved generalization. However, this worst-case penalty comes at the expense of slower training convergence and computational overhead.

In contrast, consider the inverse variational problem:
\begin{equation}
\label{eq:anti_sam}
\min_{w \in \mathbb{R}^d} \left( \min_{\|\epsilon\| \le \rho} L(w + \epsilon) \right)
\end{equation}
We refer to Equation~\eqref{eq:anti_sam} as \textbf{Anti-SAM}. Anti-SAM is an optimistic lookahead minimization: it queries whether there exists any parameter perturbation within radius $\rho$ that drastically lowers the loss.

Empirically, Anti-SAM makes the training loss plummet at extraordinary velocity. However, it exhibits a fatal pathology: it causes severe overfitting, actively seeks sharp needles (narrow, sample-specific troughs in the training landscape), and causes validation loss to stagnate or diverge.

The central research question addressed in this paper is:
\begin{quote}
\textit{Can we establish the rigorous mathematical theory governing the Anti-SAM formula, and construct methods such that validation loss decreases synchronously with training loss, retaining its rapid loss-cutting property while achieving robust generalization?}
\end{quote}

\section{Mathematical Analysis of the Anti-SAM Operator}

\subsection{Morphological Erosion and First-Order Dynamics}
Let $L: \mathbb{R}^d \to \mathbb{R}$ be continuously differentiable. The inner problem in Equation~\eqref{eq:anti_sam} is:
\begin{equation}
\epsilon^*(w) \triangleq \arg\min_{\|\epsilon\| \le \rho} L(w + \epsilon)
\end{equation}
Using a first-order Taylor expansion around $w$:
\begin{equation}
L(w + \epsilon) = L(w) + \langle \nabla L(w), \epsilon \rangle + \mathcal{O}(\|\epsilon\|^2)
\end{equation}
The optimal perturbation on the Euclidean sphere $\|\epsilon\|_2 \le \rho$ is given by Cauchy-Schwarz:
\begin{equation}
\epsilon^*(w) = -\rho \frac{\nabla L(w)}{\|\nabla L(w)\|}
\end{equation}
Substituting $\epsilon^*(w)$ into the objective yields the surrogate function:
\begin{equation}
F_\rho(w) \triangleq \min_{\|\epsilon\| \le \rho} L(w + \epsilon) \approx L(w) - \rho \|\nabla L(w)\|
\end{equation}
In continuous geometry, $F_\rho(w)$ is the \textbf{Morphological Erosion} $\mathcal{E}_\rho[L](w)$ of the function $L$ by the structuring element $B_\rho(0)$.

\begin{theorem}[Erosion Descent Velocity]
Let $L$ be $M$-smooth with non-zero gradient $\nabla L(w) \ne 0$. For any stepsize $\eta$ and radius $\rho$ satisfying $\rho > \eta \|\nabla L(w)\|$, the first-order loss reduction of the Anti-SAM surrogate strictly exceeds that of standard gradient descent:
\begin{equation}
\Delta L_{\text{Anti-SAM}}^{(1)} = -\rho \|\nabla L(w)\| < -\eta \|\nabla L(w)\|^2 = \Delta L_{\text{GD}}^{(1)}
\end{equation}
\end{theorem}
\begin{proof}
Direct from linear expansion and the condition $\rho > \eta \|\nabla L(w)\|$. Formally machine-verified in Lean 4 (\texttt{ErosionVelocity.lean}).
\end{proof}

\section{The Generalization Catastrophe: Why Anti-SAM Overfits}

\subsection{Phase-Space Caustic Collapse}
We analyze the dynamical mapping $\Phi_\rho(w) \triangleq w + \epsilon^*(w) = w - \rho \frac{\nabla L(w)}{\|\nabla L(w)\|}$.
Let $E(w) \triangleq -\rho \frac{\nabla L(w)}{\|\nabla L(w)\|}$ denote the displacement vector field.

\begin{theorem}[Phase-Space Caustic Collapse]
The Jacobian $J_E(w)$ of the Anti-SAM displacement field satisfies:
\begin{equation}
J_E(w) = -\frac{\rho}{\|\nabla L(w)\|} P_w^\perp \nabla^2 L(w)
\end{equation}
where $P_w^\perp \triangleq I - \frac{\nabla L(w) \nabla L(w)^T}{\|\nabla L(w)\|^2}$ is the orthogonal projector onto the transverse tangent space $T_w^\perp = \ker(\nabla L(w)^T)$.

The divergence of the vector field is:
\begin{equation}
\operatorname{div}(E(w)) = -\frac{\rho}{\|\nabla L(w)\|} \operatorname{Tr}_{T_w^\perp}(\nabla^2 L(w))
\end{equation}
In regions of positive transverse curvature ($\operatorname{Tr}_{T_w^\perp}(\nabla^2 L(w)) > 0$), $\operatorname{div}(E(w)) < 0$.
Consequently, by Liouville's theorem, parameter phase-space volume contracts exponentially:
\begin{equation}
\frac{d}{dt} \ln \operatorname{Vol}(\Omega_t) = -\frac{\rho}{\|\nabla L(w)\|} \operatorname{Tr}_{T_w^\perp}(\nabla^2 L(w)) < 0
\end{equation}
\end{theorem}

\begin{corollary}[Catchment Basin Dilation]
For an isolated needle of radius $r \ll \rho$, the morphological erosion operator expands its catchment basin volume from $\mathcal{O}(r^d)$ to $\mathcal{O}(\rho^d)$, amplifying its gravitational attraction by a factor of $(\rho/r)^d \to \infty$ in overparameterized dimensions $d \gg 1$.
\end{corollary}

\subsection{Finite-Sample Noise Divergence}
On finite minibatches $S$, the stochastic gradient decomposes into population signal plus orthogonal noise:
\begin{equation}
\nabla L_S(w) = \nabla L_{\mathcal{D}}(w) + \xi_S(w), \quad \mathbb{E}[\xi_S] = 0, \quad \mathbb{E}[\|\xi_S\|^2] = \operatorname{Tr}(\Sigma) \approx d \sigma^2
\end{equation}
The empirical loss decrease is:
\begin{equation}
\Delta L_S = -\rho \sqrt{\|\nabla L_{\mathcal{D}}\|^2 + \|\xi_S\|^2}
\end{equation}
However, evaluated on the true population distribution $\mathcal{D}$, the expected inner product is:
\begin{equation}
\mathbb{E}[\Delta L_{\mathcal{D}}] = -\rho \frac{\|\nabla L_{\mathcal{D}}\|^2}{\sqrt{\|\nabla L_{\mathcal{D}}\|^2 + \|\xi_S\|^2}}
\end{equation}
The Generalization Deficit scales linearly with dimension $d$:
\begin{equation}
\operatorname{Gap} \triangleq \Delta L_{\mathcal{D}} - \Delta L_S = \rho \frac{\|\xi_S\|^2}{\sqrt{\|\nabla L_{\mathcal{D}}\|^2 + \|\xi_S\|^2}} = \mathcal{O}\left(\frac{\rho d \sigma^2}{\|\nabla L_{\mathcal{D}}\|}\right)
\end{equation}

\section{The IsoAntiSAM Framework}

To solve both failure modes, we propose \textbf{IsoAntiSAM}, defined by two synergistic pillars:

\subsection{Pillar 1: The Isochoric Gauge Condition}
We require the perturbation vector field to be solenoidal (divergence-free) on the transverse manifold:
\begin{equation}
\operatorname{div}_{T^\perp}(E_{\text{iso}}) = 0 \implies \det(J_{\Phi_{\text{iso}}}) = 1
\end{equation}
By preserving phase-space volume, sharp needles cannot act as singular sinks or dissipative attractors.

\subsection{Pillar 2: Bilateral Coherence Gating}
We split each stochastic batch into two conditionally independent micro-batches $B_1, B_2 \sim \mathcal{D}$:
\begin{equation}
g_1 = \nabla L_{B_1}(w) = \nabla L_{\mathcal{D}} + \xi_1, \quad g_2 = \nabla L_{B_2}(w) = \nabla L_{\mathcal{D}} + \xi_2
\end{equation}

\begin{theorem}[Bilateral Noise Cancellation]
Under independence $\mathbb{E}[\xi_1 \xi_2^T] = 0$ and $\mathbb{E}[\xi_i] = 0$:
\begin{equation}
\mathbb{E}[\langle g_1, g_2 \rangle] = \|\nabla L_{\mathcal{D}}\|^2 \ge 0
\end{equation}
Sample noise is identically eliminated from the cross-batch inner product.
\end{theorem}
\begin{proof}
$\langle g_1, g_2 \rangle = \langle \nabla L_{\mathcal{D}} + \xi_1, \nabla L_{\mathcal{D}} + \xi_2 \rangle = \|\nabla L_{\mathcal{D}}\|^2 + \langle \nabla L_{\mathcal{D}}, \xi_1 + \xi_2 \rangle + \langle \xi_1, \xi_2 \rangle$. Taking expectations concludes the proof. Formally machine-checked in Lean 4 (\texttt{CoherentGeneralization.lean}).
\end{proof}

We define the Bilateral Coherence Gate:
\begin{equation}
\mathcal{C}(g_1, g_2) \triangleq \max\left(0, \frac{\langle g_1, g_2 \rangle}{\|g_1\| \|g_2\|}\right)
\end{equation}
The IsoAntiSAM perturbation is:
\begin{equation}
\epsilon_{\text{iso}}^*(w) \triangleq -\rho \cdot \mathcal{C}(g_1, g_2) \cdot \frac{g_1 + g_2}{\|g_1 + g_2\|}
\end{equation}

\begin{corollary}[Synchronous Descent Guarantee]
When $w$ encounters sample noise or a spurious needle, $\mathcal{C}(g_1, g_2) \to 0$, vetoing perturbation in noise directions ($\epsilon_{\text{iso}}^* = 0$). When moving along the shared data manifold, $\mathcal{C} \approx 1$, restoring the full morphological erosion velocity:
\begin{equation}
\frac{d}{dt} L_{\mathcal{D}}(w(t)) = \frac{d}{dt} L_S(w(t)) \approx -\rho \|\nabla L_{\mathcal{D}}(w(t))\| < 0
\end{equation}
\end{corollary}
